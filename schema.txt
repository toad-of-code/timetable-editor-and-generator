CREATE TABLE public.cluster_requirements (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  cluster_id uuid,
  subject_id uuid,
  CONSTRAINT cluster_requirements_pkey PRIMARY KEY (id),
  CONSTRAINT cluster_requirements_cluster_id_fkey FOREIGN KEY (cluster_id) REFERENCES public.semester_clusters(id),
  CONSTRAINT cluster_requirements_subject_id_fkey FOREIGN KEY (subject_id) REFERENCES public.subjects(id)
);
CREATE TABLE public.professor_expertise (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  professor_id uuid,
  subject_id uuid,
  preference_level integer DEFAULT 1,
  CONSTRAINT professor_expertise_pkey PRIMARY KEY (id),
  CONSTRAINT professor_expertise_professor_id_fkey FOREIGN KEY (professor_id) REFERENCES public.professors(id),
  CONSTRAINT professor_expertise_subject_id_fkey FOREIGN KEY (subject_id) REFERENCES public.subjects(id)
);
CREATE TABLE public.professors (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name text NOT NULL,
  email text UNIQUE,
  department text,
  availability jsonb DEFAULT '[]'::jsonb,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT professors_pkey PRIMARY KEY (id)
);
CREATE TABLE public.rooms (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name text NOT NULL UNIQUE,
  capacity integer NOT NULL,
  room_type text NOT NULL CHECK (room_type = ANY (ARRAY['Lecture'::text, 'Lab'::text])),
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT rooms_pkey PRIMARY KEY (id)
);
CREATE TABLE public.semester_clusters (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  batch_year integer NOT NULL,
  semester_number integer NOT NULL,
  department text DEFAULT 'IT'::text,
  is_active boolean DEFAULT true,
  CONSTRAINT semester_clusters_pkey PRIMARY KEY (id)
);
CREATE TABLE public.student_groups (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name text NOT NULL,
  group_type text NOT NULL CHECK (group_type = ANY (ARRAY['Core'::text, 'Elective'::text])),
  student_count integer NOT NULL,
  semester integer NOT NULL,
  program text NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT student_groups_pkey PRIMARY KEY (id)
);
CREATE TABLE public.subjects (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  code text NOT NULL UNIQUE,
  name text NOT NULL,
  lectures integer DEFAULT 0,
  tutorials integer DEFAULT 0,
  practicals integer DEFAULT 0,
  practical_duration integer DEFAULT 120,
  subject_type text NOT NULL CHECK (subject_type = ANY (ARRAY['Core'::text, 'Elective'::text, 'Minor'::text])),
  created_at timestamp with time zone DEFAULT now(),
  credits integer DEFAULT 4,
  CONSTRAINT subjects_pkey PRIMARY KEY (id)
);
CREATE TABLE public.timetable_slots (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  timetable_id uuid NOT NULL,
  subject_id uuid NOT NULL,
  professor_id uuid,
  room_id uuid,
  student_group_id uuid NOT NULL,
  day_of_week integer NOT NULL CHECK (day_of_week >= 1 AND day_of_week <= 7),
  start_time time without time zone NOT NULL,
  end_time time without time zone NOT NULL,
  slot_type text NOT NULL CHECK (slot_type = ANY (ARRAY['Lecture'::text, 'Tutorial'::text, 'Practical'::text])),
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT timetable_slots_pkey PRIMARY KEY (id),
  CONSTRAINT timetable_slots_timetable_id_fkey FOREIGN KEY (timetable_id) REFERENCES public.timetables(id),
  CONSTRAINT timetable_slots_subject_id_fkey FOREIGN KEY (subject_id) REFERENCES public.subjects(id),
  CONSTRAINT timetable_slots_professor_id_fkey FOREIGN KEY (professor_id) REFERENCES public.professors(id),
  CONSTRAINT timetable_slots_room_id_fkey FOREIGN KEY (room_id) REFERENCES public.rooms(id),
  CONSTRAINT timetable_slots_student_group_id_fkey FOREIGN KEY (student_group_id) REFERENCES public.student_groups(id)
);
CREATE TABLE public.timetables (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name text NOT NULL,
  academic_year text NOT NULL,
  semester integer NOT NULL,
  status text DEFAULT 'draft'::text CHECK (status = ANY (ARRAY['draft'::text, 'published'::text, 'archived'::text])),
  created_by uuid,
  created_at timestamp with time zone DEFAULT now(),
  published_at timestamp with time zone,
  lunch_start time without time zone DEFAULT '13:00:00'::time without time zone,
  lunch_end time without time zone DEFAULT '14:30:00'::time without time zone,
  CONSTRAINT timetables_pkey PRIMARY KEY (id),
  CONSTRAINT timetables_created_by_fkey FOREIGN KEY (created_by) REFERENCES auth.users(id)
);
